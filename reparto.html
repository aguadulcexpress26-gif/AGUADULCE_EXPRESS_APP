// Iniciar la app manualmente async function iniciarApp() { let codigo = localStorage.getItem('codigo_reparto'); if (!codigo) { const res = await Swal.fire({ title: '🔐 Código de Repartidor', input: 'text', inputPlaceholder: 'Ej: R001', background: '#020617', color: 'white', confirmButtonColor: '#06b6d4' }); if (res.value) { codigo = res.value.trim().toUpperCase(); localStorage.setItem('codigo_reparto', codigo); } else { return; } } // Guardar codigo globalmente window.codigo = codigo; // Cargar pedidos await cargarPedidos(); // Ocultar loader y mostrar app document.getElementById('loader').style.display = 'none'; document.getElementById('app').style.display = 'block'; console.log('✅ App iniciada con código:', codigo); } iniciarApp();
